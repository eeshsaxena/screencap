# Applied only to the x86_64 build leg via PIP_CONSTRAINT. Forces wheels with
# minos <= 11.0 for Big Sur compat. MACOSX_DEPLOYMENT_TARGET alone does not
# filter pip's wheel selection — pip reads the runner's actual OS version,
# which on macos-15-intel picks macosx_14_0 wheels and fails the minos check.

# numpy 2.0.x ships BOTH macosx_10_9_x86_64 and macosx_14_0_x86_64 cp312
# wheels. On macOS 14+ runners (CI's macos-15-intel and local Tahoe hosts)
# pip prefers the higher-targeted wheel — macosx_14_0 — which has minos=14.0
# and fails the Big Sur minos<=11.0 gate. MACOSX_DEPLOYMENT_TARGET=11.0
# doesn't override this preference because packaging.tags reads
# platform.mac_ver() at runtime, not the env var.
# Capping to <2.0 lands on numpy 1.26.4, which only ships a macosx_10_9
# cp312 x86_64 wheel — pip has no choice and the resulting bundle passes.
numpy>=1.20.0,<2.0

# av 14+ has no cp312 x86_64 wheel; 13.x ships macosx_10_13_x86_64.
av>=13.0,<14

# Matches the pyproject.toml <0.2 cap; hard-pin for deterministic resolution.
fast-gliner==0.1.12

# 5.x wheel tags not yet validated.
ctranslate2>=4.0,<5.0

# Excluded from the PyInstaller bundle (see pyinstaller/screencap.spec), but
# faster-whisper declares a hard dep so pip still resolves it. 1.20+ tags as
# macosx_13_0+; cap to 1.19.2 (last with macosx_11_0_universal2).
onnxruntime<=1.19.2
